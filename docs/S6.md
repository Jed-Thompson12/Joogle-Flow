```jsx
import { useEffect, useState } from "react";
import "./App.css";

const API = "http://127.0.0.1:8000";

function App() {
  const [pages, setPages] = useState([]);
  const [selectedPage, setSelectedPage] = useState(null);
  const [loading, setLoading] = useState(true);

  // Load pages from FastAPI
  useEffect(() => {
    loadPages();
  }, []);

  async function loadPages() {
    try {
      const response = await fetch(`${API}/api/pages`);
      const data = await response.json();

      setPages(data);

      if (data.length > 0) {
        setSelectedPage(data[0].id);
      }
    } catch (error) {
      console.error("Could not connect to Joogle Flow API:", error);
    } finally {
      setLoading(false);
    }
  }

  const currentPage = pages.find(
    (page) => page.id === selectedPage
  );

  // Create a new page
  async function createPage() {
    const newPage = {
      title: "Untitled Page",
      blocks: [
        {
          id: Date.now(),
          type: "paragraph",
          content: "",
        },
      ],
    };

    try {
      const response = await fetch(`${API}/api/pages`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify(newPage),
      });

      const createdPage = await response.json();

      setPages((oldPages) => [
        ...oldPages,
        createdPage,
      ]);

      setSelectedPage(createdPage.id);
    } catch (error) {
      console.error("Could not create page:", error);
    }
  }

  // Save a page
  async function savePage(page) {
    try {
      await fetch(`${API}/api/pages/${page.id}`, {
        method: "PUT",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          title: page.title,
          blocks: page.blocks,
        }),
      });
    } catch (error) {
      console.error("Could not save page:", error);
    }
  }

  // Update the current page
  function updatePage(changes) {
    const updatedPage = {
      ...currentPage,
      ...changes,
    };

    setPages((oldPages) =>
      oldPages.map((page) =>
        page.id === updatedPage.id
          ? updatedPage
          : page
      )
    );

    savePage(updatedPage);
  }

  // Update a block
  function updateBlock(blockId, changes) {
    const updatedBlocks = currentPage.blocks.map(
      (block) =>
        block.id === blockId
          ? { ...block, ...changes }
          : block
    );

    updatePage({
      blocks: updatedBlocks,
    });
  }

  // Add a block
  function addBlock(type) {
    const block = {
      id: Date.now(),
      type,
      content: "",
      checked: false,
    };

    updatePage({
      blocks: [
        ...currentPage.blocks,
        block,
      ],
    });
  }

  // Delete a block
  function deleteBlock(blockId) {
    updatePage({
      blocks: currentPage.blocks.filter(
        (block) => block.id !== blockId
      ),
    });
  }

  // Delete a page
  async function deletePage(pageId) {
    if (!confirm("Delete this page?")) {
      return;
    }

    try {
      await fetch(`${API}/api/pages/${pageId}`, {
        method: "DELETE",
      });

      const remainingPages = pages.filter(
        (page) => page.id !== pageId
      );

      setPages(remainingPages);

      if (remainingPages.length > 0) {
        setSelectedPage(remainingPages[0].id);
      } else {
        setSelectedPage(null);
      }
    } catch (error) {
      console.error("Could not delete page:", error);
    }
  }

  function renderBlock(block) {
    switch (block.type) {

      case "heading":
        return (
          <input
            className="block heading-block"
            value={block.content}
            placeholder="Heading"
            onChange={(e) =>
              updateBlock(block.id, {
                content: e.target.value,
              })
            }
          />
        );

      case "todo":
        return (
          <div className="todo-block">

            <input
              type="checkbox"
              checked={block.checked || false}
              onChange={(e) =>
                updateBlock(block.id, {
                  checked: e.target.checked,
                })
              }
            />

            <input
              className={
                "block todo-input " +
                (block.checked
                  ? "completed"
                  : "")
              }
              value={block.content}
              placeholder="To-do"
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() =>
                deleteBlock(block.id)
              }
            >
              ×
            </button>

          </div>
        );

      case "bullet":
        return (
          <div className="bullet-block">

            <span>•</span>

            <input
              className="block"
              value={block.content}
              placeholder="List item"
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() =>
                deleteBlock(block.id)
              }
            >
              ×
            </button>

          </div>
        );

      case "code":
        return (
          <div className="code-block-wrapper">

            <textarea
              className="code-block"
              value={block.content}
              placeholder="Write code..."
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() =>
                deleteBlock(block.id)
              }
            >
              ×
            </button>

          </div>
        );

      case "divider":
        return (
          <div className="divider-wrapper">

            <hr />

            <button
              className="delete-block"
              onClick={() =>
                deleteBlock(block.id)
              }
            >
              ×
            </button>

          </div>
        );

      default:
        return (
          <div className="paragraph-block">

            <textarea
              className="block paragraph"
              value={block.content}
              placeholder="Type something..."
              rows={1}
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() =>
                deleteBlock(block.id)
              }
            >
              ×
            </button>

          </div>
        );
    }
  }

  if (loading) {
    return (
      <div className="loading">
        Loading Joogle Flow...
      </div>
    );
  }

  return (
    <div className="app">

      {/* SIDEBAR */}

      <aside className="sidebar">

        <div className="logo">
          <div className="logo-icon">
            J
          </div>

          <span>
            Joogle Flow
          </span>
        </div>

        <button
          className="new-page"
          onClick={createPage}
        >
          + New Page
        </button>

        <nav>

          <div className="nav-item active">
            🏠 Home
          </div>

          <div className="nav-item">
            📄 Pages
          </div>

          <div className="nav-item">
            📁 Files
          </div>

          <div className="nav-item">
            ⭐ Favourites
          </div>

          <div className="nav-item">
            🗑 Trash
          </div>

        </nav>

        <div className="sidebar-pages">

          <div className="sidebar-heading">
            PAGES
          </div>

          {pages.map((page) => (

            <div
              key={page.id}
              className={
                "sidebar-page " +
                (
                  selectedPage === page.id
                    ? "selected"
                    : ""
                )
              }
              onClick={() =>
                setSelectedPage(page.id)
              }
            >
              📄 {page.title}
            </div>

          ))}

        </div>

        <div className="sidebar-bottom">

          <div className="storage-title">
            Storage
          </div>

          <div className="storage-bar">
            <div className="storage-used"></div>
          </div>

          <div className="storage-text">
            0 GB used
          </div>

        </div>

      </aside>

      {/* MAIN */}

      <main className="main">

        <header className="topbar">

          <div className="breadcrumb">
            Joogle Flow
            {currentPage &&
              ` / ${currentPage.title}`
            }
          </div>

          <div className="topbar-actions">

            <button>
              🔍
            </button>

            <button>
              ⚙️
            </button>

            <button className="profile">
              J
            </button>

          </div>

        </header>

        {/* NO PAGE */}

        {!currentPage ? (

          <div className="empty-state">

            <h1>
              Welcome to Joogle Flow
            </h1>

            <p>
              Create your first page to get started.
            </p>

            <button
              className="new-page"
              onClick={createPage}
            >
              + Create Page
            </button>

          </div>

        ) : (

          <section className="editor-page">

            <input
              className="page-title"
              value={currentPage.title}
              onChange={(e) =>
                updatePage({
                  title: e.target.value,
                })
              }
            />

            <div className="blocks">

              {currentPage.blocks.map(
                (block) => (

                  <div
                    className="block-container"
                    key={block.id}
                  >
                    {renderBlock(block)}
                  </div>

                )
              )}

            </div>

            <div className="add-block">

              <span>
                Add block:
              </span>

              <button
                onClick={() =>
                  addBlock("paragraph")
                }
              >
                Text
              </button>

              <button
                onClick={() =>
                  addBlock("heading")
                }
              >
                Heading
              </button>

              <button
                onClick={() =>
                  addBlock("todo")
                }
              >
                To-do
              </button>

              <button
                onClick={() =>
                  addBlock("bullet")
                }
              >
                Bullet
              </button>

              <button
                onClick={() =>
                  addBlock("code")
                }
              >
                Code
              </button>

              <button
                onClick={() =>
                  addBlock("divider")
                }
              >
                Divider
              </button>

            </div>

            <button
              className="delete-page"
              onClick={() =>
                deletePage(currentPage.id)
              }
            >
              Delete Page
            </button>

          </section>

        )}

      </main>

    </div>
  );
}

export default App;
```

# What Joogle Flow is now

We have moved from a static prototype to:

```text
                  Joogle Flow
                       │
                       ▼
                 React Frontend
                       │
                    HTTP API
                       │
                       ▼
                 FastAPI Backend
                       │
                       ▼
                    SQLite
                       │
                       ▼
              joogleflow.db
```

And our API now supports:

| Operation | API                      |
| --------- | ------------------------ |
| Get pages | `GET /api/pages`         |
| Get page  | `GET /api/pages/{id}`    |
| Create    | `POST /api/pages`        |
| Save      | `PUT /api/pages/{id}`    |
| Delete    | `DELETE /api/pages/{id}` |

### One thing we'll fix soon

At the moment, **every keystroke sends a database request** because we're saving immediately.

That's okay for our prototype, but we don't want that in the final Joogle Flow.

We'll eventually implement:

```text
User types
   ↓
Wait 500ms
   ↓
No more typing?
   ↓
Autosave
   ↓
✓ Saved
```
