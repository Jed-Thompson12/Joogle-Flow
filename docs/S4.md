```text
📄 My Page

┌─────────────────────────────────────┐
│ My Page                             │
│                                     │
│ Welcome to my page                  │ ← heading
│ This is some text.                  │ ← paragraph
│                                     │
│ ☑ Finish Joogle Flow                │ ← todo
│                                     │
│ • Learn React                       │ ← bullet
│ • Learn FastAPI                     │
│                                     │
│ ─────────────────────────────────── │ ← divider
│                                     │
│ Start writing...                    │
└─────────────────────────────────────┘
```

```jsx
import { useState } from "react";
import "./App.css";

function App() {
  const [pages, setPages] = useState([
    {
      id: 1,
      title: "Welcome to Joogle Flow",
      blocks: [
        {
          id: 1,
          type: "heading",
          content: "Welcome to Joogle Flow",
        },
        {
          id: 2,
          type: "paragraph",
          content: "This is your first Joogle Flow page.",
        },
        {
          id: 3,
          type: "todo",
          content: "Build the block editor",
          checked: false,
        },
      ],
    },
    {
      id: 2,
      title: "My Notes",
      blocks: [
        {
          id: 4,
          type: "paragraph",
          content: "Start writing your notes here...",
        },
      ],
    },
  ]);

  const [selectedPage, setSelectedPage] = useState(1);

  const currentPage = pages.find(
    (page) => page.id === selectedPage
  );

  const updatePage = (updatedPage) => {
    setPages(
      pages.map((page) =>
        page.id === updatedPage.id ? updatedPage : page
      )
    );
  };

  const createPage = () => {
    const newPage = {
      id: Date.now(),
      title: "Untitled Page",
      blocks: [
        {
          id: Date.now() + 1,
          type: "paragraph",
          content: "",
        },
      ],
    };

    setPages([...pages, newPage]);
    setSelectedPage(newPage.id);
  };

  const addBlock = (type) => {
    const newBlock = {
      id: Date.now(),
      type,
      content: "",
      checked: false,
    };

    updatePage({
      ...currentPage,
      blocks: [...currentPage.blocks, newBlock],
    });
  };

  const updateBlock = (blockId, changes) => {
    const updatedBlocks = currentPage.blocks.map((block) =>
      block.id === blockId
        ? { ...block, ...changes }
        : block
    );

    updatePage({
      ...currentPage,
      blocks: updatedBlocks,
    });
  };

  const deleteBlock = (blockId) => {
    updatePage({
      ...currentPage,
      blocks: currentPage.blocks.filter(
        (block) => block.id !== blockId
      ),
    });
  };

  const updateTitle = (title) => {
    updatePage({
      ...currentPage,
      title,
    });
  };

  const renderBlock = (block) => {
    switch (block.type) {
      case "heading":
        return (
          <input
            className="block heading-block"
            value={block.content}
            placeholder="Heading"
            onChange={(e) =>
              updateBlock(block.id, {
                content: e.target.value,
              })
            }
          />
        );

      case "todo":
        return (
          <div className="todo-block">
            <input
              type="checkbox"
              checked={block.checked}
              onChange={(e) =>
                updateBlock(block.id, {
                  checked: e.target.checked,
                })
              }
            />

            <input
              className={
                "block todo-input " +
                (block.checked ? "completed" : "")
              }
              value={block.content}
              placeholder="To-do"
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() => deleteBlock(block.id)}
            >
              ×
            </button>
          </div>
        );

      case "bullet":
        return (
          <div className="bullet-block">
            <span>•</span>

            <input
              className="block"
              value={block.content}
              placeholder="List item"
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() => deleteBlock(block.id)}
            >
              ×
            </button>
          </div>
        );

      case "divider":
        return (
          <div className="divider-wrapper">
            <hr />
            <button
              className="delete-block"
              onClick={() => deleteBlock(block.id)}
            >
              ×
            </button>
          </div>
        );

      case "code":
        return (
          <div className="code-block-wrapper">
            <textarea
              className="code-block"
              value={block.content}
              placeholder="Write code..."
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() => deleteBlock(block.id)}
            >
              ×
            </button>
          </div>
        );

      default:
        return (
          <div className="paragraph-block">
            <textarea
              className="block paragraph"
              value={block.content}
              placeholder="Type something..."
              rows={1}
              onChange={(e) =>
                updateBlock(block.id, {
                  content: e.target.value,
                })
              }
            />

            <button
              className="delete-block"
              onClick={() => deleteBlock(block.id)}
            >
              ×
            </button>
          </div>
        );
    }
  };

  return (
    <div className="app">

      <aside className="sidebar">

        <div className="logo">
          <div className="logo-icon">J</div>
          <span>Joogle Flow</span>
        </div>

        <button
          className="new-page"
          onClick={createPage}
        >
          + New Page
        </button>

        <nav>
          <div className="nav-item active">
            🏠 Home
          </div>

          <div className="nav-item">
            📄 Pages
          </div>

          <div className="nav-item">
            📁 Files
          </div>

          <div className="nav-item">
            ⭐ Favourites
          </div>

          <div className="nav-item">
            🗑 Trash
          </div>
        </nav>

        <div className="sidebar-pages">

          <div className="sidebar-heading">
            PAGES
          </div>

          {pages.map((page) => (
            <div
              key={page.id}
              className={
                "sidebar-page " +
                (selectedPage === page.id
                  ? "selected"
                  : "")
              }
              onClick={() =>
                setSelectedPage(page.id)
              }
            >
              📄 {page.title}
            </div>
          ))}

        </div>

        <div className="sidebar-bottom">

          <div className="storage-title">
            Storage
          </div>

          <div className="storage-bar">
            <div className="storage-used"></div>
          </div>

          <div className="storage-text">
            0 GB used
          </div>

        </div>

      </aside>

      <main className="main">

        <header className="topbar">

          <div className="breadcrumb">
            Joogle Flow / {currentPage.title}
          </div>

          <div className="topbar-actions">
            <button>🔍</button>
            <button>⚙️</button>
            <button className="profile">
              J
            </button>
          </div>

        </header>

        <section className="editor-page">

          <input
            className="page-title"
            value={currentPage.title}
            onChange={(e) =>
              updateTitle(e.target.value)
            }
          />

          <div className="blocks">

            {currentPage.blocks.map((block) => (
              <div
                className="block-container"
                key={block.id}
              >
                {renderBlock(block)}
              </div>
            ))}

          </div>

          <div className="add-block">

            <span>Add block:</span>

            <button
              onClick={() => addBlock("paragraph")}
            >
              Text
            </button>

            <button
              onClick={() => addBlock("heading")}
            >
              Heading
            </button>

            <button
              onClick={() => addBlock("todo")}
            >
              To-do
            </button>

            <button
              onClick={() => addBlock("bullet")}
            >
              Bullet
            </button>

            <button
              onClick={() => addBlock("code")}
            >
              Code
            </button>

            <button
              onClick={() => addBlock("divider")}
            >
              Divider
            </button>

          </div>

        </section>

      </main>

    </div>
  );
}

export default App;
```
