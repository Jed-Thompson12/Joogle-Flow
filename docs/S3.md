Delete everything in that file and replace it with:

```jsx
import { useState } from "react";
import "./App.css";

function App() {
  const [pages, setPages] = useState([
    { id: 1, title: "Welcome to Joogle Flow" },
    { id: 2, title: "My Notes" },
  ]);

  const [selectedPage, setSelectedPage] = useState(1);

  const createPage = () => {
    const newPage = {
      id: Date.now(),
      title: "Untitled Page",
    };

    setPages([...pages, newPage]);
    setSelectedPage(newPage.id);
  };

  const currentPage = pages.find((page) => page.id === selectedPage);

  return (
    <div className="app">

      <aside className="sidebar">

        <div className="logo">
          <div className="logo-icon">J</div>
          <span>Joogle Flow</span>
        </div>

        <button className="new-page" onClick={createPage}>
          + New Page
        </button>

        <nav>
          <div className="nav-item active">
            🏠 Home
          </div>

          <div className="nav-item">
            📄 Pages
          </div>

          <div className="nav-item">
            📁 Files
          </div>

          <div className="nav-item">
            ⭐ Favourites
          </div>

          <div className="nav-item">
            🗑 Trash
          </div>
        </nav>

        <div className="sidebar-bottom">
          <div className="storage-title">
            Storage
          </div>

          <div className="storage-bar">
            <div className="storage-used"></div>
          </div>

          <div className="storage-text">
            0 GB used
          </div>
        </div>

      </aside>

      <main className="main">

        <header className="topbar">

          <div className="breadcrumb">
            Joogle Flow
          </div>

          <div className="topbar-actions">
            <button>🔍</button>
            <button>⚙️</button>
            <button className="profile">J</button>
          </div>

        </header>

        <section className="content">

          <div className="welcome">
            <p className="small-text">
              YOUR WORKSPACE
            </p>

            <h1>
              Welcome to Joogle Flow
            </h1>

            <p className="description">
              Your personal workspace for documents, files and ideas.
            </p>
          </div>

          <div className="section-header">
            <h2>Recent Pages</h2>

            <button onClick={createPage}>
              + New Page
            </button>
          </div>

          <div className="page-grid">

            {pages.map((page) => (

              <div
                key={page.id}
                className={
                  "page-card " +
                  (selectedPage === page.id ? "selected" : "")
                }
                onClick={() => setSelectedPage(page.id)}
              >

                <div className="page-icon">
                  📄
                </div>

                <div className="page-info">

                  <div className="page-title">
                    {page.title}
                  </div>

                  <div className="page-date">
                    Just now
                  </div>

                </div>

              </div>

            ))}

          </div>

          {currentPage && (

            <div className="editor">

              <input
                className="editor-title"
                value={currentPage.title}
                onChange={(e) => {
                  setPages(
                    pages.map((page) =>
                      page.id === currentPage.id
                        ? {
                            ...page,
                            title: e.target.value,
                          }
                        : page
                    )
                  );
                }}
              />

              <textarea
                className="editor-body"
                placeholder="Start writing..."
              />

            </div>

          )}

        </section>

      </main>

    </div>
  );
}

export default App;
```
