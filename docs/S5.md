**Step 5: make Joogle Flow actually save things.**

Right now, if you refresh the browser, your pages disappear because they're only stored in React memory.

We're going to change:

```text
React
  ↓
FastAPI
  ↓
SQLite
```

So Joogle Flow has **persistent pages**.

Run:

```powershell

```

Paste:

```python
from sqlalchemy import create_engine
from sqlalchemy.orm import declarative_base, sessionmaker

DATABASE_URL = "sqlite:///./joogleflow.db"

engine = create_engine(
    DATABASE_URL,
    connect_args={"check_same_thread": False}
)

SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine
)

Base = declarative_base()
```

Save it.

# 5.3 Create the Page model

Run:

```powershell
notepad models.py
```

Paste:

```python
from sqlalchemy import Column, Integer, String, Text
from database import Base


class Page(Base):
    __tablename__ = "pages"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String(255), nullable=False)
    content = Column(Text, nullable=False, default="[]")
```

Save it.

### Why `content` is text

Our blocks look like:

```json
[
  {
    "id": 1,
    "type": "heading",
    "content": "Welcome"
  },
  {
    "id": 2,
    "type": "paragraph",
    "content": "Hello!"
  }
]
```

We'll store that block structure as JSON inside SQLite.

Later, when we build the full database, we'll make the architecture more sophisticated.

# 5.4 Replace `main.py`

Run:

```powershell
notepad main.py
```

Replace everything with:

```python
import json

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

from database import Base, SessionLocal, engine
from models import Page


Base.metadata.create_all(bind=engine)

app = FastAPI(title="Joogle Flow API")


class PageCreate(BaseModel):
    title: str
    blocks: list


class PageUpdate(BaseModel):
    title: str
    blocks: list


@app.get("/")
def root():
    return {
        "name": "Joogle Flow",
        "status": "online"
    }


@app.get("/api/pages")
def get_pages():
    db = SessionLocal()

    try:
        pages = db.query(Page).all()

        return [
            {
                "id": page.id,
                "title": page.title,
                "blocks": json.loads(page.content)
            }
            for page in pages
        ]

    finally:
        db.close()


@app.get("/api/pages/{page_id}")
def get_page(page_id: int):
    db = SessionLocal()

    try:
        page = db.query(Page).filter(
            Page.id == page_id
        ).first()

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        return {
            "id": page.id,
            "title": page.title,
            "blocks": json.loads(page.content)
        }

    finally:
        db.close()


@app.post("/api/pages")
def create_page(page_data: PageCreate):
    db = SessionLocal()

    try:
        page = Page(
            title=page_data.title,
            content=json.dumps(page_data.blocks)
        )

        db.add(page)
        db.commit()
        db.refresh(page)

        return {
            "id": page.id,
            "title": page.title,
            "blocks": page_data.blocks
        }

    finally:
        db.close()


@app.put("/api/pages/{page_id}")
def update_page(
    page_id: int,
    page_data: PageUpdate
):
    db = SessionLocal()

    try:
        page = db.query(Page).filter(
            Page.id == page_id
        ).first()

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        page.title = page_data.title
        page.content = json.dumps(page_data.blocks)

        db.commit()

        return {
            "id": page.id,
            "title": page.title,
            "blocks": page_data.blocks
        }

    finally:
        db.close()


@app.delete("/api/pages/{page_id}")
def delete_page(page_id: int):
    db = SessionLocal()

    try:
        page = db.query(Page).filter(
            Page.id == page_id
        ).first()

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        db.delete(page)
        db.commit()

        return {
            "success": True
        }

    finally:
        db.close()
```

Save and close it.
