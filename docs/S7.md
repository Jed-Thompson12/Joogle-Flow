# 7.2 — Replace `models.py`

Open:

```text
C:\JoogleFlow\backend\models.py
```

Replace everything with:

```python
from sqlalchemy import Column, Integer, String, Text, ForeignKey
from database import Base


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    username = Column(String(100), unique=True, nullable=False, index=True)
    password_hash = Column(String(255), nullable=False)


class Page(Base):
    __tablename__ = "pages"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)

    title = Column(String(255), nullable=False)
    content = Column(Text, nullable=False, default="[]")
```

Now the database relationship is:

```text
User
 │
 ├── Page
 ├── Page
 ├── Page
 └── Page
```

# 7.3 — Replace `main.py`

Open:

```text
C:\JoogleFlow\backend\main.py
```

Replace the entire file with:

```python
import json
from datetime import datetime, timedelta, timezone

from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import jwt, JWTError
from pydantic import BaseModel

from pwdlib import PasswordHash

from database import Base, SessionLocal, engine
from models import User, Page


# --------------------------------------------------
# DATABASE
# --------------------------------------------------

Base.metadata.create_all(bind=engine)


# --------------------------------------------------
# APP
# --------------------------------------------------

app = FastAPI(title="Joogle Flow API")


app.add_middleware(
    CORSMiddleware,
    allow_origins=[
        "http://localhost:5173",
        "http://127.0.0.1:5173",
    ],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


# --------------------------------------------------
# AUTHENTICATION
# --------------------------------------------------

password_hash = PasswordHash.recommended()

SECRET_KEY = "CHANGE-THIS-LATER"
ALGORITHM = "HS256"

security = HTTPBearer()


# --------------------------------------------------
# REQUEST MODELS
# --------------------------------------------------

class RegisterRequest(BaseModel):
    username: str
    password: str


class LoginRequest(BaseModel):
    username: str
    password: str


class PageCreate(BaseModel):
    title: str
    blocks: list


class PageUpdate(BaseModel):
    title: str
    blocks: list


# --------------------------------------------------
# PASSWORD FUNCTIONS
# --------------------------------------------------

def hash_password(password: str):
    return password_hash.hash(password)


def verify_password(password: str, hashed_password: str):
    return password_hash.verify(password, hashed_password)


# --------------------------------------------------
# JWT FUNCTIONS
# --------------------------------------------------

def create_token(user_id: int):
    expires = datetime.now(timezone.utc) + timedelta(days=7)

    payload = {
        "sub": str(user_id),
        "exp": expires,
    }

    return jwt.encode(
        payload,
        SECRET_KEY,
        algorithm=ALGORITHM
    )


def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security)
):
    token = credentials.credentials

    try:
        payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM]
        )

        user_id = payload.get("sub")

        if user_id is None:
            raise HTTPException(
                status_code=401,
                detail="Invalid token"
            )

        return int(user_id)

    except (JWTError, ValueError):
        raise HTTPException(
            status_code=401,
            detail="Invalid or expired token"
        )


# --------------------------------------------------
# BASIC
# --------------------------------------------------

@app.get("/")
def root():
    return {
        "name": "Joogle Flow",
        "status": "online"
    }


# --------------------------------------------------
# REGISTER
# --------------------------------------------------

@app.post("/api/register")
def register(user_data: RegisterRequest):

    username = user_data.username.strip()

    if len(username) < 3:
        raise HTTPException(
            status_code=400,
            detail="Username must be at least 3 characters"
        )

    if len(user_data.password) < 8:
        raise HTTPException(
            status_code=400,
            detail="Password must be at least 8 characters"
        )

    db = SessionLocal()

    try:

        existing_user = (
            db.query(User)
            .filter(User.username == username)
            .first()
        )

        if existing_user:
            raise HTTPException(
                status_code=400,
                detail="Username already exists"
            )

        user = User(
            username=username,
            password_hash=hash_password(user_data.password)
        )

        db.add(user)
        db.commit()
        db.refresh(user)

        token = create_token(user.id)

        return {
            "token": token,
            "username": user.username
        }

    finally:
        db.close()


# --------------------------------------------------
# LOGIN
# --------------------------------------------------

@app.post("/api/login")
def login(user_data: LoginRequest):

    db = SessionLocal()

    try:

        user = (
            db.query(User)
            .filter(User.username == user_data.username)
            .first()
        )

        if not user:
            raise HTTPException(
                status_code=401,
                detail="Incorrect username or password"
            )

        if not verify_password(
            user_data.password,
            user.password_hash
        ):
            raise HTTPException(
                status_code=401,
                detail="Incorrect username or password"
            )

        token = create_token(user.id)

        return {
            "token": token,
            "username": user.username
        }

    finally:
        db.close()


# --------------------------------------------------
# CURRENT USER
# --------------------------------------------------

@app.get("/api/me")
def get_me(user_id: int = Depends(get_current_user)):

    db = SessionLocal()

    try:

        user = (
            db.query(User)
            .filter(User.id == user_id)
            .first()
        )

        if not user:
            raise HTTPException(
                status_code=404,
                detail="User not found"
            )

        return {
            "id": user.id,
            "username": user.username
        }

    finally:
        db.close()


# --------------------------------------------------
# GET PAGES
# --------------------------------------------------

@app.get("/api/pages")
def get_pages(
    user_id: int = Depends(get_current_user)
):

    db = SessionLocal()

    try:

        pages = (
            db.query(Page)
            .filter(Page.user_id == user_id)
            .all()
        )

        return [
            {
                "id": page.id,
                "title": page.title,
                "blocks": json.loads(page.content)
            }
            for page in pages
        ]

    finally:
        db.close()


# --------------------------------------------------
# GET ONE PAGE
# --------------------------------------------------

@app.get("/api/pages/{page_id}")
def get_page(
    page_id: int,
    user_id: int = Depends(get_current_user)
):

    db = SessionLocal()

    try:

        page = (
            db.query(Page)
            .filter(
                Page.id == page_id,
                Page.user_id == user_id
            )
            .first()
        )

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        return {
            "id": page.id,
            "title": page.title,
            "blocks": json.loads(page.content)
        }

    finally:
        db.close()


# --------------------------------------------------
# CREATE PAGE
# --------------------------------------------------

@app.post("/api/pages")
def create_page(
    page_data: PageCreate,
    user_id: int = Depends(get_current_user)
):

    db = SessionLocal()

    try:

        page = Page(
            user_id=user_id,
            title=page_data.title,
            content=json.dumps(page_data.blocks)
        )

        db.add(page)
        db.commit()
        db.refresh(page)

        return {
            "id": page.id,
            "title": page.title,
            "blocks": page_data.blocks
        }

    finally:
        db.close()


# --------------------------------------------------
# UPDATE PAGE
# --------------------------------------------------

@app.put("/api/pages/{page_id}")
def update_page(
    page_id: int,
    page_data: PageUpdate,
    user_id: int = Depends(get_current_user)
):

    db = SessionLocal()

    try:

        page = (
            db.query(Page)
            .filter(
                Page.id == page_id,
                Page.user_id == user_id
            )
            .first()
        )

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        page.title = page_data.title
        page.content = json.dumps(page_data.blocks)

        db.commit()

        return {
            "id": page.id,
            "title": page.title,
            "blocks": page_data.blocks
        }

    finally:
        db.close()


# --------------------------------------------------
# DELETE PAGE
# --------------------------------------------------

@app.delete("/api/pages/{page_id}")
def delete_page(
    page_id: int,
    user_id: int = Depends(get_current_user)
):

    db = SessionLocal()

    try:

        page = (
            db.query(Page)
            .filter(
                Page.id == page_id,
                Page.user_id == user_id
            )
            .first()
        )

        if not page:
            raise HTTPException(
                status_code=404,
                detail="Page not found"
            )

        db.delete(page)
        db.commit()

        return {
            "success": True
        }

    finally:
        db.close()
```


The backend is now doing this:

```python
.filter(Page.user_id == user_id)
```

So if another account logs in, it gets **only its own pages**.
